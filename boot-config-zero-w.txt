# ── Raspberry Pi Zero W (BCM2835) low-power config ───────────────
# Pi Zero 2W version: boot-config.txt. Key differences for the Zero W:
#   - core_freq must stay >= 250: on the BCM2835 the SD card host clock
#     is derived from core_freq, and values below 250 can cause SD
#     corruption. The Zero 2W's BCM2710 auto-clamps the SD clock to 500,
#     which is why boot-config.txt can safely use 200.
#   - led0/led1 dtparams omitted: the Zero W's single green LED is a
#     hardwired power LED, not GPIO-controllable.

# ── Clock & voltage ──────────────────────────────────────────────
arm_freq=600
core_freq=250
core_freq_min=250
over_voltage=-3
over_voltage_min=-3
temp_limit=65

# ── Disable unused peripherals ───────────────────────────────────
dtparam=uart0=off
dtparam=i2c0=off
dtparam=i2c1=off
dtparam=spi=off
dtparam=audio=off
dtoverlay=disable-bt
camera_auto_detect=0

# ── Keep ─────────────────────────────────────────────────────────
# USB:  (default, on)
# WiFi: (default, on — brcmfmac stays loaded)

# ── Misc ─────────────────────────────────────────────────────────
loglevel=1
quiet

# ── DHT ambient sensor (iio method) ──────────────────────────────
# Userspace bit-banging (Adafruit/blinka) is unreliable on the Zero W's
# single-core BCM2835. Use the in-kernel dht11 IIO driver instead:
# uncomment below and set "dht_method": "iio" in config.json.
# The pin must match "dht_gpio". The modules don't always autoload from
# the overlay, so also load them at boot:
#   sudo modprobe industrialio && sudo modprobe dht11
#   printf 'industrialio\ndht11\n' | sudo tee /etc/modules-load.d/dht11.conf
# Verify after reboot with:
#   dmesg | grep -i dht    and    ls /sys/bus/iio/devices/
# Requires kernel >= 6.6 with CONFIG_IIO_DHT11 (modules industrialio + dht11).
# dtoverlay=dht11,gpio_pin=4
